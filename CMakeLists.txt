find_program(PSP_OBJCOPY
    NAMES psp-objcopy
    PATHS "$ENV{PSPDEV}/bin"
    REQUIRED
)
# ── Embed assets into the ELF  ───────────────────
file(GLOB_RECURSE GAME_ASSETS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/assets/*")
list(FILTER GAME_ASSETS EXCLUDE REGEX "/\\.[^/]*$")

set(EMBEDDED_ASSET_DIR "${CMAKE_BINARY_DIR}/embedded_assets")
file(MAKE_DIRECTORY "${EMBEDDED_ASSET_DIR}")

set(EMBEDDED_ASSET_OBJECTS "")
set(ASSET_EXTERN_DECLS     "")
set(ASSET_TABLE_ENTRIES    "")

foreach(ASSET_FILE ${GAME_ASSETS})
    file(RELATIVE_PATH ASSET_REL "${CMAKE_SOURCE_DIR}/assets" "${ASSET_FILE}")

    string(REGEX REPLACE "[^A-Za-z0-9]" "_" ASSET_LABEL "asset_${ASSET_REL}")

    set(ASSET_OBJ "${EMBEDDED_ASSET_DIR}/${ASSET_LABEL}.o")

    string(REGEX REPLACE "[^A-Za-z0-9]" "_" ASSET_SYMBOL_PATH "${ASSET_FILE}")

    add_custom_command(
        OUTPUT "${ASSET_OBJ}"
        COMMAND "${PSP_OBJCOPY}"
            -I binary
            -O elf32-littlemips
            -B mips
            --rename-section .data=.rodata,alloc,load,readonly,data,contents
            "${ASSET_FILE}"
            "${ASSET_OBJ}"
        COMMAND "${PSP_OBJCOPY}"
            --redefine-sym
            "_binary_${ASSET_SYMBOL_PATH}_start=${ASSET_LABEL}"
            --redefine-sym
            "_binary_${ASSET_SYMBOL_PATH}_end=${ASSET_LABEL}_end"
            --strip-symbol
            "_binary_${ASSET_SYMBOL_PATH}_size"
            "${ASSET_OBJ}"
        DEPENDS "${ASSET_FILE}"
        COMMENT "Embedding assets/${ASSET_REL}"
        VERBATIM
    )
    #message(STATUS "OBJ SYMBOL: _binary_${ASSET_SYMBOL_PATH}_start")

    list(APPEND EMBEDDED_ASSET_OBJECTS "${ASSET_OBJ}")
    string(APPEND ASSET_EXTERN_DECLS
        "extern const unsigned char ${ASSET_LABEL}[];\nextern const unsigned char ${ASSET_LABEL}_end[];\n")
    string(APPEND ASSET_TABLE_ENTRIES
        "    { \"${ASSET_REL}\", ${ASSET_LABEL}, ${ASSET_LABEL}_end },\n")
endforeach()

file(WRITE "${CMAKE_BINARY_DIR}/assets.c"
"/* AUTO-GENERATED by CMakeLists.txt */
#include \"assets.h\"
#include <string.h>

${ASSET_EXTERN_DECLS}
typedef struct {
    const char           *name;
    const unsigned char  *start;
    const unsigned char  *end;
} AssetTableEntry;

static const AssetTableEntry assets[] = {
${ASSET_TABLE_ENTRIES}};
#define ASSET_COUNT (sizeof(assets) / sizeof(assets[0]))

EmbeddedAsset asset_find(const char *name) {
    for (size_t i = 0; i < ASSET_COUNT; i++) {
        if (strcmp(assets[i].name, name) == 0) {
            EmbeddedAsset a;
            a.data = assets[i].start;
            a.size = (size_t)(assets[i].end - assets[i].start);
            return a;
        }
    }
    EmbeddedAsset none = { NULL, 0 };
    return none;
}
")

set_source_files_properties(${EMBEDDED_ASSET_OBJECTS} PROPERTIES
    EXTERNAL_OBJECT TRUE
    GENERATED       TRUE)
target_sources(${PROJECT_NAME} PRIVATE
    ${EMBEDDED_ASSET_OBJECTS}
    "${CMAKE_BINARY_DIR}/assets.c")
